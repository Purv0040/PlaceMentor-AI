Great question! That's a classic Google & Bloomberg follow-up. Here are the two standard approaches interviewers look for:

palette 1. DFS 3-State Coloring

White (0 = Unvisited), Gray (1 = Currently In Recursion Stack), Black (2 = Fully Explored). Hitting a Gray node immediately proves a back-edge and identifies cycle membership.

filter_vintage 2. Kahn's In-Degree Drain

Any vertices retaining in-degree > 0 after queue exhaustion are mathematically part of or downstream dependencies of the deadlock cycle.

// State 0: Unvisited, 1: Visiting (Gray), 2: Visited (Black)
bool reconstructCycleDFS(int u, const vectorint>>& adj, vector<int>& state, vector<int>& parent, vector<int>& cycle) {
    state[u] = 1; // Mark Gray
    for (int v : adj[u]) {
        if (state[v] == 1) {
            // Cycle found! Backtrack along parent chain to reconstruct
            cycle.push_back(v);
            for (int curr = u; curr != v; curr = parent[curr]) {
                cycle.push_back(curr);
            }
            cycle.push_back(v);
            reverse(cycle.begin(), cycle.end());
            return true;
        } else if (state[v] == 0) {
            parent[v] = u;
            if (reconstructCycleDFS(v, adj, state, parent, cycle)) return true;
        }
    }
    state[u] = 2; // Mark Black
    return false;
}
